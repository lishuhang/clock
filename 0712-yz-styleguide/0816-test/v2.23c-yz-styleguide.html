RECORDING SCRIPT (Python + Playwright + ffmpeg)

import asyncio, subprocess, os from playwright.async_api import async_playwright async def record_mp4(html_path: str, mp4_path: str): """Record a 1:1 animated HTML to 16s MP4.""" FRAMES_DIR = "/tmp/yz_frames" FPS, ENTRANCE_DUR, STATIC_DUR = 30, 6.0, 4.0 os.makedirs(FRAMES_DIR, exist_ok=True) for f in os.listdir(FRAMES_DIR): os.remove(os.path.join(FRAMES_DIR, f)) async with async_playwright() as p: browser = await p.chromium.launch() page = await browser.new_page( viewport={"width": 1100, "height": 1100}, device_scaleFactor=1 ) await page.goto(f"file://{html_path}", wait_until="domcontentloaded") await page.evaluate("() => document.fonts.ready") await page.wait_for_timeout(500) # let animation start container = await page.query_selector(".chart-container-1x1") total_frames = int(FPS * ENTRANCE_DUR) for i in range(total_frames): await container.screenshot( path=f"{FRAMES_DIR}/frame_{i:04d}.png") await asyncio.sleep(1.0 / FPS) await browser.close() last_frame = f"{FRAMES_DIR}/frame_{total_frames-1:04d}.png" entrance = f"{FRAMES_DIR}/entrance.mp4" static = f"{FRAMES_DIR}/static.mp4" exit_rev = f"{FRAMES_DIR}/exit.mp4" # Step 1: entrance subprocess.run([ "ffmpeg", "-y", "-framerate", str(FPS), "-i", f"{FRAMES_DIR}/frame_%04d.png", "-c:v", "libx264", "-pix_fmt", "yuv420p", "-preset", "ultrafast", entrance ]) # Step 2: static (last frame loop) subprocess.run([ "ffmpeg", "-y", "-loop", "1", "-i", last_frame, "-t", str(STATIC_DUR), "-r", str(FPS), "-c:v", "libx264", "-pix_fmt", "yuv420p", "-preset", "ultrafast", static ]) # Step 3: exit (reverse entrance) subprocess.run([ "ffmpeg", "-y", "-i", entrance, "-vf", "reverse", "-c:v", "libx264", "-pix_fmt", "yuv420p", "-preset", "ultrafast", exit_rev ]) # Step 4: concat concat_file = f"{FRAMES_DIR}/concat.txt" with open(concat_file, "w") as f: f.write(f"file '{entrance}'\n") f.write(f"file '{static}'\n") f.write(f"file '{exit_rev}'\n") subprocess.run([ "ffmpeg", "-y", "-f", "concat", "-safe", "0", "-i", concat_file, "-c:v", "libx264", "-pix_fmt", "yuv420p", "-preset", "ultrafast", mp4_path ]) print(f"Done: {mp4_path}")

CORE RULES

  1. Auto-play: CSS animation triggers on page load, no JS trigger needed
  2. 4-stage hierarchy: logo(0s) → title(1.5s) → chrome(3s) → body(4.5s)
  3. 16s total: 6s entrance + 4s hold + 6s exit (reverse)
  4. Element screenshot: record .chart-container-1x1, not viewport
  5. 3-step ffmpeg: entrance → static(loop) → exit(reverse) → concat
  6. Chart-specific: pie sweep, bar grow, table row stagger, card slideUp
  7. Clean recording: hide banner, white background, no padding
